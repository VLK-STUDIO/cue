---

Create the Cue environment once, then define an overlay from it:

<Demo>
  <DemoPreview>
    <DefaultDemo />
  </DemoPreview>

```tsx
import { createCue } from "@vlkoss/cue";
import { Button } from "@/components/ui/button";
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogFooter,
  DialogTitle,
} from "@/components/ui/dialog";
import { CueBackdrop } from "@/components/cue-backdrop";

export const cue = createCue({ backdrop: CueBackdrop });

export const settingsDialog = cue.createOverlay((_props, ctx) => (
  <Dialog open={ctx.open} onOpenChange={ctx.onOpenChange}>
    <DialogContent showBackdrop={false}>
      <DialogTitle>Settings</DialogTitle>
      <DialogDescription>No local useState required.</DialogDescription>
      <DialogFooter>
        <Button variant="outline" onClick={() => ctx.close()}>
          Close
        </Button>
      </DialogFooter>
    </DialogContent>
  </Dialog>
));

settingsDialog.open();
```

</Demo>

The same `cue` instance supplies `cue.OverlayProvider` near the app root. `open()` returns the close function for the new instance. `closeAll()` closes every live instance of this overlay definition.
